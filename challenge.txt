    function updateLocksUI() {
      unlockedLocksCount = parseInt(localStorage.getItem('unlockedLocksCount') || '0', 10);

      for (let i = 1; i <= 4; i++) {
        const card = document.getElementById(`lockCard${i}`);
        const icon = document.getElementById(`lockIcon${i}`);
        const progress = document.getElementById(`lockProgress${i}`);

        if (i <= unlockedLocksCount) {
          card.classList.remove('active-lock');
          card.classList.add('unlocked-lock');
          icon.textContent = '🔑';
          progress.textContent = 'تم الفتح! 🔑';
        } else if (i === unlockedLocksCount + 1) {
          card.classList.remove('unlocked-lock');
          card.classList.add('active-lock');
          icon.textContent = '🔒';
          progress.textContent = '0 / 4 أسئلة';
        } else {
          card.classList.remove('active-lock', 'unlocked-lock');

function nextIntroStep() {
  const dialogueText = document.getElementById('introDialogueText');

        { q: "تتشابه بعض العظام بين الإنسان والحيوانات في...؟", opts: ["الوظيفة والتركيب العام", "الحجم تماماً", "اللون الداخلي"], ans: 0 }
      ]
    };

    function openLockQuiz(lockNum) {
      if (lockNum > unlockedLocksCount + 1) {
        showSuccessModal(
          "🔒 قفل مغلق!",
          "يرجى حل القفل السابق أولاً للحصول على مفتاحه!",
          false
        );
        return;
      }
      if (lockNum <= unlockedLocksCount) {
        showSuccessModal(
          "🔑 قفل مفتوح!",
          "لقد فتحتِ هذا القفل وحصلتِ على مفتاحه بالفعل! 🎉",
          false
        );
        return;
      }

      currentLock = lockNum;
      currentQuestionIndex = 0;
      showQuestion();
      document.getElementById('quizModal').style.display = 'flex';
    }

    function showQuestion() {
      isAnswering = false;
      const feedbackMsg = document.getElementById('feedbackMsg');
      feedbackMsg.className = 'feedback-msg';
      feedbackMsg.style.display = 'none';

      const questions = lockQuizzes[currentLock];
      const qData = questions[currentQuestionIndex];

      document.getElementById('quizTitle').textContent = `تحدي القفل ${currentLock} (السؤال ${currentQuestionIndex + 1} من 4)`;
      document.getElementById('quizQuestionText').textContent = qData.q;

      currentShuffledOptions = qData.opts.map((opt, idx) => ({
        text: opt,
        isCorrect: idx === qData.ans
      }));

      currentShuffledOptions.sort(() => Math.random() - 0.5);

      const optsContainer = document.getElementById('optionsContainer');
      optsContainer.innerHTML = '';

      currentShuffledOptions.forEach((optObj, index) => {
        const btn = document.createElement('button');
        btn.className = 'option-btn';
        btn.id = `optBtn_${index}`;
        btn.textContent = optObj.text;
        btn.onclick = () => checkAnswer(optObj.isCorrect, btn);
        optsContainer.appendChild(btn);
      });
    }

    function checkAnswer(isCorrect, selectedBtn) {
      if (isAnswering) return;
      isAnswering = true;

      const feedbackMsg = document.getElementById('feedbackMsg');

      if (!isCorrect) {
        playSFX('wrong.mp3');
        selectedBtn.classList.add('wrong-btn');
        feedbackMsg.textContent = "❌ إجابة خاطئة! حاولي مرة أخرى بتركيز.";
        feedbackMsg.className = "feedback-msg error-msg";

        setTimeout(() => {
          selectedBtn.classList.remove('wrong-btn');
          isAnswering = false;
        }, 1200);

        return;
      }

      playSFX('correct.mp3');
      selectedBtn.classList.add('correct-btn');

      setTimeout(() => {
        currentQuestionIndex++;

        if (currentQuestionIndex < lockQuizzes[currentLock].length) {
          showQuestion();
        } else {
          unlockedLocksCount = Math.max(unlockedLocksCount, currentLock);
          localStorage.setItem('unlockedLocksCount', String(unlockedLocksCount));
          updateLocksUI();

          document.getElementById('quizModal').style.display = 'none';

          if (unlockedLocksCount < 4) {
            showSuccessModal(
              "🔑 تم فتح القفل بنجاح!",
              `🎉 رائع جداً! لقد حصلتِ على المفتاح رقم (${currentLock}). القفل التالي أصبح متاحاً الآن!`,
              false
            );
          } else {
            playSFX('succes.mp3');
            showSuccessModal(
              "🏆 مبروك يا أبطال! تم الهروب!",
              "🥳 رائعة جداً! لقد جمعتِ المفاتيح الأربعة وتم فتح باب المعمل بنجاح والهروب بنجاح تام!",
              true
            );
          }
        }
      }, 1500);
    }

    function showSuccessModal(title, text, isFinal) {
      document.getElementById('successTitle').textContent = title;
      document.getElementById('successText').textContent = text;
      document.getElementById('successModal').style.display = 'flex';

      setTimeout(() => {
        document.getElementById('successModal').style.display = 'none';
        if (isFinal) {
          localStorage.removeItem('unlockedLocksCount');
          goHome();
        }
      }, 2500);
    }

    function goHome() {
      window.location.href = 'levels.html';
    }
